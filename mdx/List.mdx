import OlItem from './OlItem'
import UlItem from './UlItem'

await function List({ items, ordered = false }: { items: string[]; emptyText?: string; ordered?: boolean }) {
  await new Promise((res) => setTimeout(res, 100));

  return (
    {items.length === 0 ? 'Empty' : items.map((item, index) => ordered ? <OlItem item={item} index={index} /> : <UlItem item={item} />)}
  )
}
