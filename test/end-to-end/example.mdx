import { getData } from "./api/getData" // import external TS functions

import { Dashboard } from "./Dashboard"; // import reusable components

async function TestComponent() { // async function support
    const { data, error, timedout } = await getData(); // call await in the component

    if (error) return false; // falsy values don't take up space on the line
    if (timedout) return (**API Error**) // one-line markdown return
    if (!data) {
        return (
            **Error**: No data available

            {{ null }} // returning null removes the previously rendered line (if empty)
        )
    }

    // Multi-line markdown
    // Indentation support
    // Conditional rendering
    // XML support
    // Components
    // Empty lines remain
    return (
        # Admin panel
        {{ data.isAuthorized ? (
            Authorized
        ): (
            Not Authorized
        )}}
        {{!data.active && (
            Account is inactive
        )}}
        - Name: {{ data.name }}
        - Description: {{ data.description }}
          Access your information here
        
        <content> 
        <@Dashboard />
        </content>

    )
}
